import React, { useState } from 'react';
import { View, Text, Button, FlatList, StyleSheet } from 'react-native';

export default function App() {
  const [cart, setCart] = useState([]);
  const menu = [
    { id: 1, name: 'Pizza 🍕', price: 12 },
    { id: 2, name: 'Burger 🍔', price: 8 },
    { id: 3, name: 'Sushi 🍣', price: 15 },
  ];

  const addToCart = (item) => {
    setCart([...cart, item]);
  };

  const total = cart.reduce((sum, item) => sum + item.price, 0);

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Commande de nourriture</Text>
      
      <FlatList
        data={menu}
        keyExtractor={(item) => item.id.toString()}
        renderItem={({ item }) => (
          <View style={styles.item}>
            <Text>{item.name} - {item.price}€</Text>
            <Button title="Ajouter" onPress={() => addToCart(item)} />
          </View>
        )}
      />

      <Text style={styles.total}>Panier : {cart.length} article(s) | Total : {total}€</Text>
      <Button title="Passer la commande" disabled={cart.length === 0} />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, backgroundColor: '#f5f5f5' },
  title: { fontSize: 24, fontWeight: 'bold', marginBottom: 20 },
  item: { padding: 15, backgroundColor: 'white', marginBottom: 10, borderRadius: 8 },
  total: { marginTop: 20, fontSize: 18, fontWeight: 'bold' },
});
